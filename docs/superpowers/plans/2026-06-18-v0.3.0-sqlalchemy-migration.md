# v0.3.0 — SQLAlchemy 1.x → 2.0 Migration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship one new Refactron transform — `sqlalchemy_query_to_select` — that auto-rewrites the safe subset of SQLAlchemy 1.x `session.query(...)` chains to the 2.0 `session.execute(select(...))` form, flags every unsafe pattern with a structured precondition record, and surfaces query-level test coverage so the existing 3-gate verifier's "tests pass" claim actually means something on data-layer code.

**Architecture:** Three new pieces compose around Refactron's existing engines. (1) A tree-sitter **detector** classifies every query-rooted chain as safe-shape / flag-shape. (2) A LibCST **sidecar** rewrites safe-shape chains and emits a precondition per flag-shape chain. (3) A new **Python line-coverage reporter** runs `coverage.py` once per analyze invocation, parses `coverage json`, and tags every finding with `testCovered: yes | no | unknown`. The reporter degrades gracefully when `coverage.py` isn't installed in the user's project.

**Tech Stack:** TypeScript (tree-sitter-python via existing adapter, ts-morph for nothing here, Vitest for tests). Python sidecar (LibCST, stdlib only). Shell-out to user's Python for the coverage reporter (`coverage.py` is the user's dep, not Refactron's). Atomic-writer, blast-radius, 3-gate verifier, precondition discipline — all existing, all reused.

```typescript
import type { SyntaxNode } from 'tree-sitter';
import type { DetectorContext, DetectorFinding } from '../types.js';
import { register } from '../index.js';

const REMEDIATION = 25;

/** Walk the call chain leftward from a tail call node. Returns the head
 *  attribute (e.g. `session.query`) and the list of method names in order. */
function walkChain(call: SyntaxNode): { head: SyntaxNode | null; methods: string[] } {
  const methods: string[] = [];
  let cur: SyntaxNode | null = call;
  let head: SyntaxNode | null = null;
  while (cur && cur.type === 'call') {
    const fn = cur.childForFieldName('function');
    if (!fn || fn.type !== 'attribute') break;
    const attr = fn.childForFieldName('attribute');
    if (attr) methods.push(attr.text);
    const recv = fn.childForFieldName('object');
    if (recv && recv.type === 'call') {
      cur = recv;
      continue;
    }
    head = fn;
    break;
  }
  methods.reverse(); // walker went right→left; restore source order
  return { head, methods };
}

/** True iff the head attribute resolves to `<name>.query` — i.e. a method call
 *  named `query` on some receiver. We don't try to prove the receiver is a real
 *  SQLAlchemy Session; that's the user's responsibility via `--transforms=`. */
function isQueryHead(head: SyntaxNode | null): boolean {
  if (!head || head.type !== 'attribute') return false;
  const attr = head.childForFieldName('attribute');
  return attr?.text === 'query';
}

export function detect(ctx: DetectorContext): DetectorFinding[] {
  const findings: DetectorFinding[] = [];
  let counter = 0;
  function visit(node: SyntaxNode): void {
    if (node.type === 'call') {
      // Only treat as a candidate when this call is the OUTER call of a chain
      // (its parent is not itself a function-attribute of another call).
      const parent = node.parent;
      const isInner =
        parent?.type === 'attribute' &&
        parent.parent?.type === 'call' &&
        parent.parent.childForFieldName('function') === parent;
      if (!isInner) {
        const { head } = walkChain(node);
        if (isQueryHead(head)) {
          findings.push({
            id: `sql-${ctx.relPath}-${node.startPosition.row}-${counter++}`,
            file: ctx.relPath,
            line: node.startPosition.row + 1,
            transformId: 'sqlalchemy_query_to_select',
            remediationMinutes: REMEDIATION,
            confidence: 'medium',
            // eslint-disable-next-line @typescript-eslint/no-explicit-any
            meta: { shape: 'safe' } as any,
          });
        }
      }
    }
    for (const c of node.namedChildren) visit(c);
  }
  visit(ctx.tree.rootNode);
  return findings;
}

register({ transformId: 'sqlalchemy_query_to_select', lang: 'python', detect });
```

```typescript
function classifyMeta(
  headCall: SyntaxNode, // the `session.query(...)` call node
  methods: string[],
): { shape: 'safe' } | { shape: 'flag'; flagReason: string } {
  // Head args — flag multi-column / multi-entity at the head.
  const args = headCall.childForFieldName('arguments');
  if (args) {
    const named = args.namedChildren.filter((c) => c.type !== 'comment');
    if (named.length > 1) return { shape: 'flag', flagReason: 'multi-column-select' };
    if (named.length === 1 && named[0]!.type === 'attribute') {
      return { shape: 'flag', flagReason: 'multi-column-select' };
    }
  }
  // Method chain — flag specific shapes.
  if (methods.includes('options')) {
    return { shape: 'flag', flagReason: 'joinedload-needs-unique' };
  }
  if (methods.includes('update')) {
    return { shape: 'flag', flagReason: 'bulk-update-semantics' };
  }
  if (methods.includes('delete')) {
    return { shape: 'flag', flagReason: 'bulk-delete-semantics' };
  }
  if (methods.includes('select_entity_from')) {
    return { shape: 'flag', flagReason: 'select-entity-from' };
  }
  if (methods.filter((m) => m === 'join').length >= 3) {
    return { shape: 'flag', flagReason: 'complex-joins' };
  }
  return { shape: 'safe' };
}
```

In `detect()`, replace `meta: { shape: 'safe' }` with the call to `classifyMeta(node, methods)` (and capture `methods` from `walkChain`).

```typescript
import { spawn } from 'node:child_process';
import * as fs from 'node:fs/promises';
import * as os from 'node:os';
import * as path from 'node:path';

export interface CoverageReportInput {
  projectRoot: string;
  testCmd?: string; // default: 'pytest -q'
  pythonBin?: string; // default: 'python3'
  _probeOverride?: boolean; // test-only injection point
}

export interface CoverageReport {
  coverageToolFound: boolean;
  coveredLines: Set<string>; // `${relPath}:${line}` (1-indexed)
  runDurationMs: number;
}

/** Probe whether `coverage.py` is importable in the user's Python. */
function probeCoverage(pythonBin: string): Promise<boolean> {
  return new Promise((resolve) => {
    const p = spawn(pythonBin, ['-c', 'import coverage'], { stdio: 'ignore' });
    p.on('exit', (code) => resolve(code === 0));
    p.on('error', () => resolve(false));
  });
}

/** Run a command in `cwd` with `env` and resolve to its exit code + stderr.
 *  Returns null exit code on spawn error. */
function runCmd(
  cmd: string,
  args: string[],
  cwd: string,
  env: NodeJS.ProcessEnv,
): Promise<{ code: number | null; stderr: string }> {
  return new Promise((resolve) => {
    const p = spawn(cmd, args, { cwd, env, stdio: ['ignore', 'pipe', 'pipe'] });
    let stderr = '';
    p.stderr.on('data', (d: Buffer) => {
      stderr += d.toString();
    });
    p.stdout.on('data', () => {
      /* drop */
    });
    p.on('exit', (code) => resolve({ code, stderr }));
    p.on('error', () => resolve({ code: null, stderr }));
  });
}

export async function reportCoverage(input: CoverageReportInput): Promise<CoverageReport> {
  const t0 = performance.now();
  const pythonBin = input.pythonBin ?? 'python3';
  const testCmd = input.testCmd ?? 'pytest -q';

  const found = input._probeOverride ?? (await probeCoverage(pythonBin));
  if (!found) {
    return { coverageToolFound: false, coveredLines: new Set(), runDurationMs: 0 };
  }

  const tmp = await fs.mkdtemp(path.join(os.tmpdir(), 'refactron-cov-'));
  const dataFile = path.join(tmp, '.coverage');
  const jsonFile = path.join(tmp, 'coverage.json');
  const env = { ...process.env, COVERAGE_FILE: dataFile };

  // 1) Run tests under coverage.
  const runArgs = ['-m', 'coverage', 'run', '--data-file', dataFile, '-m', ...testCmd.split(' ')];
  await runCmd(pythonBin, runArgs, input.projectRoot, env);

  // 2) Emit JSON.
  await runCmd(
    pythonBin,
    ['-m', 'coverage', 'json', '--data-file', dataFile, '-o', jsonFile],
    input.projectRoot,
    env,
  );

  // 3) Parse.
  const covered = new Set<string>();
  try {
    const raw = await fs.readFile(jsonFile, 'utf8');
    const parsed = JSON.parse(raw) as { files?: Record<string, { executed_lines?: number[] }> };
    for (const [relPath, fileData] of Object.entries(parsed.files ?? {})) {
      for (const line of fileData.executed_lines ?? []) {
        covered.add(`${relPath}:${line}`);
      }
    }
  } catch {
    // JSON missing/unparseable → return empty but still mark tool as found.
  }

  await fs.rm(tmp, { recursive: true, force: true });

  return {
    coverageToolFound: true,
    coveredLines: covered,
    runDurationMs: performance.now() - t0,
  };
}
```

```python
"""Chain extractor for SQLAlchemy `session.query(...).filter(...).all()` shapes.

Used by `sqlalchemy_query_to_select.py` (the sidecar) to walk a tail call
node leftward and produce a structured shape: head call (e.g. `query(M)`),
the receiver before `query` (e.g. `session`), and the ordered method chain.
"""
from __future__ import annotations
from dataclasses import dataclass
from typing import List, Sequence, Tuple
import libcst as cst


@dataclass
class ChainShape:
    head_call: cst.Call          # the `session.query(...)` Call node
    head_recv: cst.BaseExpression  # the receiver of `.query` (typically a Name `session`)
    methods: List[Tuple[str, Sequence[cst.Arg]]]  # ordered method links AFTER `query`


def extract_chain(call: cst.Call) -> ChainShape:
    """Walk leftward from `call` (which is the rightmost terminal call, e.g.
    `.all()`), collecting method names + args until we reach the `query(...)` head.

    Raises ValueError if the chain is not query-rooted."""
    methods: List[Tuple[str, Sequence[cst.Arg]]] = []
    cur: cst.BaseExpression = call

    while isinstance(cur, cst.Call):
        func = cur.func
        if not isinstance(func, cst.Attribute):
            raise ValueError("chain root is not an attribute call")
        method_name = func.attr.value
        if method_name == "query":
            # Found the head.
            return ChainShape(
                head_call=cur,
                head_recv=func.value,
                methods=list(reversed(methods)),
            )
        methods.append((method_name, cur.args))
        cur = func.value

    raise ValueError("chain does not terminate at .query(...)")
```

```python
from typing import Optional

def resolve_model(head_call: cst.Call) -> Optional[cst.BaseExpression]:
    """Returns the single model class expression iff `head_call` is
    `query(<Name>)`. Returns None otherwise (multi-entity, attribute-form
    indicating multi-column, or any computed expression)."""
    args = head_call.args
    if len(args) != 1:
        return None
    arg = args[0]
    # Refuse keyword / starred / unpacked args.
    if arg.keyword is not None or arg.star not in ("",):
        return None
    val = arg.value
    if isinstance(val, cst.Name):
        return val
    return None
```

```python
"""Refactron transform: SQLAlchemy 1.x query → 2.0 select.

Phase-1 scope: rewrite the safe subset only. Every refusal path emits a
precondition record with id + reason. The transform never crashes — parse
errors surface via `ok=False, error=...`.
"""
from __future__ import annotations
import os
import sys

sys.path.insert(0, os.path.dirname(__file__))
from _base import read_source, emit  # noqa: E402
from _sqlalchemy_chain import extract_chain, resolve_model  # noqa: E402

import libcst as cst  # noqa: E402


class QueryRewriter(cst.CSTTransformer):
    def __init__(self) -> None:
        self.preconditions: list[dict] = []
        self.changed = False

    def leave_Call(
        self, original: cst.Call, updated: cst.Call
    ) -> cst.BaseExpression:
        # Only act on the OUTER call of a chain. We detect that by checking
        # the parent isn't itself a `func` for a wrapping Call — but LibCST
        # transformers don't expose parents, so we instead check whether
        # `updated` is itself an attribute call (.foo()) where the receiver
        # is itself a call whose chain ends in .query(). If so, treat as outer.
        try:
            shape = extract_chain(updated)
        except ValueError:
            return updated

        # Phase 1 of the rewrite — only handle simple .filter() + .all().
        method_names = [m for m, _ in shape.methods]
        if not (len(method_names) == 2
                and method_names[0] == "filter"
                and method_names[1] == "all"):
            return updated  # later tasks will extend
        if resolve_model(shape.head_call) is None:
            return updated  # multi-entity head — Task 14 flags this

        # Rebuild: session.execute(select(<Model>).where(<cond>)).scalars().all()
        model_arg = shape.head_call.args[0].value
        filter_args = shape.methods[0][1]  # the .filter(...) args

        select_call = cst.Call(
            func=cst.Name("select"),
            args=[cst.Arg(value=model_arg)],
        )
        where_call = cst.Call(
            func=cst.Attribute(value=select_call, attr=cst.Name("where")),
            args=list(filter_args),
        )
        execute_call = cst.Call(
            func=cst.Attribute(value=shape.head_recv, attr=cst.Name("execute")),
            args=[cst.Arg(value=where_call)],
        )
        scalars_call = cst.Call(
            func=cst.Attribute(value=execute_call, attr=cst.Name("scalars")),
            args=[],
        )
        all_call = cst.Call(
            func=cst.Attribute(value=scalars_call, attr=cst.Name("all")),
            args=[],
        )
        self.changed = True
        return all_call


def main() -> None:
    if len(sys.argv) < 2:
        emit(ok=False, error="usage: sqlalchemy_query_to_select.py <file>")
        return
    src = read_source(sys.argv[1])
    try:
        module = cst.parse_module(src)
    except cst.ParserSyntaxError as e:
        emit(ok=False, error=f"parse error: {e}")
        return

    visitor = QueryRewriter()
    new_module = module.visit(visitor)

    if not visitor.changed:
        emit(ok=True, new_content="", preconditions=visitor.preconditions)
        return
    emit(ok=True, new_content=new_module.code, preconditions=visitor.preconditions)


if __name__ == "__main__":
    main()
```

```python
def _has_import(module: cst.Module, mod: str, name: str) -> bool:
    import libcst.matchers as m
    for node in m.findall(module, m.ImportFrom()):
        if isinstance(node.module, cst.Name) and node.module.value == mod:
            if isinstance(node.names, cst.ImportStar):
                return True
            for alias in node.names:
                if isinstance(alias.name, cst.Name) and alias.name.value == name:
                    return True
    return False


def _inject_imports(module: cst.Module, names: list[str]) -> cst.Module:
    """Insert `from sqlalchemy import <names_combined>` after the last
    `from __future__ import` line, deduped against existing imports."""
    needed = [n for n in names if not _has_import(module, "sqlalchemy", n)]
    if not needed:
        return module

    new_import = cst.SimpleStatementLine(body=[
        cst.ImportFrom(
            module=cst.Name("sqlalchemy"),
            names=[cst.ImportAlias(name=cst.Name(n)) for n in sorted(needed)],
        )
    ])

    body = list(module.body)
    # Find last __future__ line.
    insert_at = 0
    for i, stmt in enumerate(body):
        if isinstance(stmt, cst.SimpleStatementLine):
            for sub in stmt.body:
                if (isinstance(sub, cst.ImportFrom)
                    and isinstance(sub.module, cst.Name)
                    and sub.module.value == "__future__"):
                    insert_at = i + 1
    body.insert(insert_at, new_import)
    return module.with_changes(body=body)
```

Add a per-`FunctionDef` visitor that tracks assignments. If a variable is bound to `session.query(M)` (directly or transitively) and then reassigned to its own filtered self, OR if `.all()`/`.first()`/etc is called on a variable that came from a prior `session.query(...)` assignment, emit `multi-statement-chain`:

```python
class MultiStmtChainVisitor(cst.CSTVisitor):
    def __init__(self) -> None:
        self.preconditions: list[dict] = []
    def visit_FunctionDef(self, node: cst.FunctionDef) -> None:
        # Track names bound to session.query(...) or to themselves via .filter() etc.
        query_bound: set[str] = set()
        for stmt in (node.body.body if isinstance(node.body, cst.IndentedBlock) else []):
            if isinstance(stmt, cst.SimpleStatementLine):
                for sub in stmt.body:
                    if isinstance(sub, cst.Assign) and len(sub.targets) == 1:
                        tgt = sub.targets[0].target
                        if isinstance(tgt, cst.Name) and _is_query_rooted(sub.value):
                            query_bound.add(tgt.value)
                    # Reassignment pattern: q = q.filter(...) where q is query-bound.
                    if (isinstance(sub, cst.Assign) and len(sub.targets) == 1
                        and isinstance(sub.value, cst.Call)
                        and isinstance(sub.value.func, cst.Attribute)
                        and isinstance(sub.value.func.value, cst.Name)
                        and sub.value.func.value.value in query_bound):
                        target = sub.targets[0].target
                        if isinstance(target, cst.Name) and target.value in query_bound:
                            self.preconditions.append({
                                "id": "multi-statement-chain",
                                "satisfied": False,
                                "reason": "Query built across multiple statements with "
                                          "conditional appends. Auto-rewrite cannot prove "
                                          "the assembled WHERE clause is equivalent.",
                            })
                            return  # one flag per function
```
